# Stop and remove all containers
docker rm -f $(docker ps -aq) 2>/dev/null

# Remove all Docker images
docker rmi -f $(docker images -aq) 2>/dev/null

# Remove all Docker volumes (THIS DELETES DATABASE DATA)
docker volume rm $(docker volume ls -q) 2>/dev/null

# Remove all Docker networks except defaults
docker network rm $(docker network ls -q | grep -v "bridge\|host\|none") 2>/dev/null

# Full Docker system cleanup
docker system prune -af --volumes

# See logs
docker compose logs -f (web/api/postgres)

pnpm dlx shadcn@latest add <component> -c web
